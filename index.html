微积分：六类必考题

范围是一元函数微积分：极限与连续、导数、中值定理与导数应用、不定积分、定积分及应用。

  1. 求极限。顺序：化简 → 等价无穷小（只替换乘除因子）→ 洛必达（每步先确认是 0/0 或 ∞/∞）→ 泰勒（加减项用）。1∞ 型用 lim fg = elim (f−1)g。
  2. 求导。分段点必须用定义并分左右；幂指函数用对数求导；参数方程二阶导 d²y/dx² = [d(dy/dx)/dt] ÷ (dx/dt)，别漏除 dx/dt。
  3. 中值定理证明。把 ξ 换成 x，凑成某个函数的导数，对它用罗尔定理（辅助函数见下表）。
  4. 导数应用。极值、最值、凹凸与拐点、渐近线；证不等式就设 F = 左 − 右，求导判号。
  5. 积分计算。先试凑微分；根式用三角代换；分部积分按“反对幂三指”选 u；定积分换元必须换限，对称区间先看奇偶。
  6. 变限积分与应用。d/dx ∫aφ(x) f(t)dt = f(φ(x))·φ′(x)，常与洛必达合出；面积、旋转体体积先画图。

线性代数：五类必考题

主线：行列式 → 矩阵 → 方程组 → 特征值 → 二次型，“秩”贯穿始终。课程含解析几何的话，另补向量积、平面与直线方程。

  1. 行列式计算。用性质化上三角；行和相等时各列加到第一列再提公因子；范德蒙德直接套公式。
  2. 求逆与矩阵方程。(A | E) → (E | A⁻¹)；AX = B 用 (A | B) → (E | X)。乘法不可交换，注意左乘右乘。
  3. 含参数方程组。增广矩阵化行阶梯，按 r(A) 与 r(A|b) 讨论无解 / 唯一解 / 无穷多解；通解 = 特解 + 齐次通解，基础解系含 n − r(A) 个向量。
  4. 相似对角化。解 |λE − A| = 0 求特征值，再解 (λE − A)x = 0 求特征向量；有 n 个线性无关特征向量才可对角化。实对称矩阵要正交化、单位化。
  5. 二次型。写矩阵时交叉项系数除以 2；正交变换化标准形就是第 4 类题；正定 ⇔ 顺序主子式全 > 0 ⇔ 特征值全 > 0。

struct Node { int data; Node *next; };

// 尾插法建表：输入若干整数，以 0 结束
Node* create() {
    Node *head = nullptr, *tail = nullptr; int x;
    while (cin >> x && x != 0) {
        Node *p = new Node; p->data = x; p->next = nullptr;
        if (head == nullptr) head = p; else tail->next = p;
        tail = p;
    }
    return head;
}

// 遍历
void print(Node *head) {
    for (Node *p = head; p != nullptr; p = p->next) cout << p->data << ' ';
}

// 在升序链表中插入，保持有序
Node* insert(Node *head, int x) {
    Node *s = new Node; s->data = x;
    if (head == nullptr || x < head->data) { s->next = head; return s; }
    Node *p = head;
    while (p->next != nullptr && p->next->data < x) p = p->next;
    s->next = p->next;   // 先接后面
    p->next = s;         // 再接前面，顺序不能反
    return head;
}

// 删除第一个值为 x 的结点
Node* del(Node *head, int x) {
    if (head == nullptr) return head;
    if (head->data == x) { Node *t = head; head = head->next; delete t; return head; }
    Node *p = head;
    while (p->next != nullptr && p->next->data != x) p = p->next;
    if (p->next != nullptr) { Node *t = p->next; p->next = t->next; delete t; }
    return head;
}
// 冒泡排序（升序）
void bubble(int a[], int n) {
    for (int i = 0; i < n - 1; i++)
        for (int j = 0; j < n - 1 - i; j++)
            if (a[j] > a[j + 1]) { int t = a[j]; a[j] = a[j + 1]; a[j + 1] = t; }
}

// 选择排序（升序）
void selectSort(int a[], int n) {
    for (int i = 0; i < n - 1; i++) {
        int k = i;
        for (int j = i + 1; j < n; j++) if (a[j] < a[k]) k = j;
        if (k != i) { int t = a[i]; a[i] = a[k]; a[k] = t; }
    }
}

// 二分查找（a 已升序），找不到返回 -1
int binSearch(int a[], int n, int x) {
    int lo = 0, hi = n - 1;
    while (lo <= hi) {
        int mid = (lo + hi) / 2;
        if (a[mid] == x) return mid;
        if (a[mid] < x) lo = mid + 1; else hi = mid - 1;
    }
    return -1;
}

int gcd(int a, int b) { return b == 0 ? a : gcd(b, a % b); }

// 汉诺塔：把 n 个盘从 a 经 b 移到 c
void hanoi(int n, char a, char b, char c) {
    if (n == 1) { cout << a << "->" << c << endl; return; }
    hanoi(n - 1, a, c, b);
    cout << a << "->" << c << endl;
    hanoi(n - 1, b, a, c);
}

// 大纲样题：输入 n 个学生（学号、姓名、成绩），按成绩降序，输出前三名
#include <iostream>
#include <string>
using namespace std;

struct Student { int id; string name; double score; };

void sortDesc(Student s[], int n) {
    for (int i = 0; i < n - 1; i++)
        for (int j = 0; j < n - 1 - i; j++)
            if (s[j].score < s[j + 1].score) { Student t = s[j]; s[j] = s[j + 1]; s[j + 1] = t; }
}

int main() {
    int n; cin >> n;
    Student *s = new Student[n];          // 动态内存分配
    for (int i = 0; i < n; i++) cin >> s[i].id >> s[i].name >> s[i].score;
    sortDesc(s, n);
    for (int i = 0; i < 3 && i < n; i++)    // 别忘了 n < 3 的情况
        cout << s[i].id << " " << s[i].name << " " << s[i].score << endl;
    delete[] s;                            // new[] 对应 delete[]
    return 0;
}

陷阱清单

  • 5/2 是 2，5/2.0 才是 2.5
  • if (a = 0) 是赋值，条件恒为假
  • char a[] = "abc"; 占 4 个字节（含 '\0'）
  • 形参 int a[] 实为指针，sizeof(a) 不是数组大小
  • &&、|| 短路：右边的 i++ 可能不执行
  • static 局部变量只初始化一次，多次调用会累积
  • 引用参数 int &a 会改到实参，值参数不会
  • switch 没有 break 会一直往下执行
  • for(...); 末尾多一个分号，循环体就是空的
  • 删除链表结点前先保存 next；new[] 配 delete[]
